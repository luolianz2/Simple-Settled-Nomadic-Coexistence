# 游牧传统树（非游牧帝国专用）

# 开门：游牧（启动"逐星"局势）
tr_nomadic_legacy_adopt = {
	possible = {
		is_nomadic = no
	}
	# 常驻国家修正：全帝国方舟宜居 +100%
	modifier = {
		pc_ark_habitability = 1.0
	}
	on_enabled = {
		# 直接启动逐星局势
		start_situation = {
			type = settled_arkship_building
		}
	}
	custom_tooltip_with_modifiers = tr_nomadic_legacy_adopt_effects
}

# 右1：运行储备（解锁法令-炽燃）
tr_nomadic_legacy_1 = {
	possible = {
		is_nomadic = no
	}
	custom_tooltip = tr_nomadic_legacy_1_effects
}

# 右2：飞往无垠（每月凝聚力+10%、闪避+5、亚光速航速+10%）（需先点右1运行储备）
tr_nomadic_legacy_2 = {
	possible = {
		is_nomadic = no
		has_tradition = tr_nomadic_legacy_1
	}
	modifier = {
		country_unity_produces_mult = 0.1
		ship_evasion_add = 5
		ship_speed_mult = 0.1
	}
}

# 左1：移动补给站（采集资源+20%）

tr_nomadic_legacy_5 = {
